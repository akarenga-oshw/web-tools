基板側のスケッチ 
#include "RTC.h"

volatile bool tick = false;
void onTick() { tick = true; }

char buf[16];
uint8_t len = 0;

void setup() {
  Serial.begin(115200);
  RTC.begin();
  RTC.setPeriodicCallback(onTick, Period::ONCE_EVERY_1_SEC);
}

void loop() {
  // 1秒ごとに現在時刻を出力する(照合用)
  if (tick) {
    tick = false;
    RTCTime now;
    RTC.getTime(now);
    Serial.println(now.getUnixTime());
  }

  // T<epoch> を受け取ったら RTC を合わせる(ブロックしない)
  while (Serial.available()) {
    char c = Serial.read();
    if (c == '\n' || c == '\r') {
      buf[len] = '\0';
      if (len > 1 && buf[0] == 'T') {
        RTCTime t;
        t.setUnixTime(strtoul(buf + 1, nullptr, 10));
        RTC.setTime(t);
      }
      len = 0;
    } else if (len < sizeof(buf) - 1) {
      buf[len++] = c;
    }
  }
}

Serial.find() と parseInt() は待ちが入って1秒ごとの出力を乱すので、受信は自前のバッファに変えています。「ローカル(JST)のエポック」を選ぶと RTC には JST がそのまま入ります。RTC を UTC で回すなら「UTC エポック」を選んでください。
